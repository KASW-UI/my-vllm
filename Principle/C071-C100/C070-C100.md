```python
# routing.py
class MoERouter(nn.Module):
    def __init__(self, hidden_dim, num_experts, top_k=2):
        self.gate = nn.Linear(hidden_dim, num_experts, bias=False)
        self.top_k = top_k

    def forward(self, x):
        router_logits = self.gate(x)         # [num_tokens, num_experts]
        routing_weights, expert_indices = torch.topk(router_logits, self.top_k, dim=-1)
        routing_weights = F.softmax(routing_weights, dim=-1)
        return routing_weights, expert_indices
```

```python
# model_executor/layers/fused_moe/layer.py
class MoELayer(nn.Module):
    def __init__(self, hidden_dim, intermediate_dim, num_experts=8, top_k=2):
        self.router = MoERouter(hidden_dim, num_experts, top_k)
        self.experts = nn.ModuleList([
            ExpertMLP(hidden_dim, intermediate_dim) for _ in range(num_experts)

    def forward(self, x):

        output = fused_moe(x, self.experts, routing_weights, expert_indices, self.top_k)
        return output
```

```python
class GroupCoordinator:
    def __init__(self, group, device, world_size, rank):
        self.group = group
        self.device = device
        self.world_size = world_size
        self.rank = rank

    def all_reduce(self, tensor):

            return tensor
        torch.distributed.all_reduce(tensor, group=self.group)
        return tensor

            return tensor

            return tensor
        torch.distributed.broadcast(tensor, src=src, group=self.group)
        return tensor
```

```python
class CPGroup:
    """Context Parallel group: 序列切分 + ring 通信。
    
    通信模式:
    - 每个 GPU 持有: Q_local [batch, seqlen//cp_size, heads, head_dim]

        ...

        self.prev_rank = (cp_rank - 1) % cp_size
```

```
拓扑布局示例 (8 GPU):
  TP=2, PP=2, EP=2:
    GPU 0,1: stage 0, expert_group 0 (TP=2)
    GPU 2,3: stage 1, expert_group 0 (TP=2)
    GPU 4,5: stage 0, expert_group 1 (TP=2)
    GPU 6,7: stage 1, expert_group 1 (TP=2)

  CP=2, TP=2:
    GPU 0,1: CP rank 0 (TP=2)
    GPU 2,3: CP rank 1 (TP=2)

通信开销模型:
  TP: O(hidden_dim × seqlen) per layer → 高频小量
  PP: O(hidden_dim × seqlen) per stage → 低频大量
  EP: O(top_k × hidden_dim × seqlen) per MoE layer → 偶发
  CP: O(num_heads × head_dim × seqlen/cp_size) per ring step → 批量中等

兼容矩阵:
  TP × PP: ✅ (每个 PP stage 内部做 TP)
  TP × EP: ✅ (每个 EP rank 内部做 TP)
  TP × CP: ✅ (每个 CP rank 内部做 TP)
  PP × CP: ⚠️ (需要跨 stage 同步 ring state)
  EP × CP: ⚠️ (All-to-All 与 ring 的通信重叠复杂)
```

验证：TP=2 + PP=2 端到端运行 OPT-350m。

```python
class FP8LinearMethod:
    def process_weights_after_loading(self, layer):
        """加载权重后转换为 FP8 格式，缓存 per-channel scale。"""
        weight_fp32 = layer.weight.data.float()
        amax = weight_fp32.abs().max(dim=1, keepdim=True).values
        scale = amax / 448.0  # E4M3 max
        layer.weight_scale = scale
        layer.weight.data = (weight_fp32 / scale).to(torch.float8_e4m3fn)

    def apply(self, layer, x, bias=None):
        x_scale = x.abs().max() / 448.0
        x_fp8 = (x / x_scale).to(torch.float8_e4m3fn)
        output = torch.matmul(
            x_fp8.to(torch.float16),
            layer.weight.data.to(torch.float16),
        )
        output = output * (x_scale * layer.weight_scale.squeeze(-1))
        if bias is not None:
            output += bias
        return output
```

```python
# v1/executor/ray_utils.py
import ray

@ray.remote(num_gpus=1)
class RayWorker:
    """Ray actor: 封装一个 Worker（GPU 进程）。
    
    生命周期:
    - __init__: 加载模型到 GPU
    - execute_model: 接收 SchedulerOutput，返回 ModelRunnerOutput
    - shutdown: 释放 GPU 资源
    """
    def __init__(self, vllm_config, rank, world_size):
        self.worker = Worker(vllm_config, rank, world_size)
        self.worker.init_device()
        self.worker.load_model()

    def execute_model(self, scheduler_output):
        return self.worker.execute_model(scheduler_output)

    def shutdown(self):
        ...

class RayPlacementGroup:
    """GPU topology-aware placement:
    
    - STRICT_SPREAD: 每个 node 最多 1 个 actor → 跨 node 分布
    - STRICT_PACK: 所有 actor 在同一 node → 低延迟通信
    - PACK: 尽可能在同一 node，超出则 spill → 默认
    """
    ...

class RayExecutor:
    """Ray 版 Executor，替换 MultiprocExecutor。
    
    Executor(ABC)
      ├── UniProcExecutor              # 单进程 (C032)
      ├── MultiprocExecutor            # 多进程 (C050)
      └── RayExecutor                  # C091: Ray actor (本 commit)
    """
    def __init__(self, vllm_config, placement_group):
        self.workers = [
            RayWorker.remote(vllm_config, rank, world_size)
            for rank in range(world_size)
        ]

    def execute_model(self, scheduler_output):
        futures = [w.execute_model.remote(scheduler_output)
                   for w in self.workers]
        return ray.get(futures)
```

核心理解点：
- **Ray actor vs multiprocessing.Process**: Ray 提供跨 node、自动故障恢复、资源调度
- **PlacementGroup**: GPU 拓扑感知——控制 actor 在哪些 GPU/node 上运行
- **何时用 Ray 何时用 multiprocess**: 单 node 用 multiprocess（低延迟），多 node 用 Ray（自动调度）
- **Ray 与 Runtime 的解耦**: RayExecutor 实现 Executor ABC，EngineCore 不感知

参考：vLLM `vllm/v1/engine/utils.py` (CoreEngineActorManager), `vllm/executor/ray_executor.py`

核心文件：
```yaml
# deploy/k8s/deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: my-vllm-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: my-vllm
  template:
    spec:
      nodeSelector:
        accelerator: nvidia-a100    # GPU node affinity
      containers:
      - name: my-vllm
        image: my-vllm:latest
        resources:
          limits:
            nvidia.com/gpu: 1       # 每个 pod 1 张 GPU
        env:
        - name: MODEL_NAME
          value: "opt-125m"
        - name: TP_SIZE
          value: "2"
        ports:
        - containerPort: 8000
        readinessProbe:
          httpGet:
            path: /health
            port: 8000
          initialDelaySeconds: 30   # 等待模型加载
          periodSeconds: 10

---
# deploy/k8s/hpa.yaml
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: my-vllm-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: my-vllm-deployment
  minReplicas: 2
  maxReplicas: 8
  metrics:
  - type: Pods
    pods:
      metric:
        name: request_queue_depth     # 自定义指标: 等待队列长度
      target:
        type: AverageValue
        averageValue: "5"              # 队列长度 > 5 → scale up
  behavior:
    scaleUp:
      stabilizationWindowSeconds: 30  # 30s 内不重复 scale
    scaleDown:
      stabilizationWindowSeconds: 300 # 5min 才 scale down
```

核心理解点：
- **Deployment spec**: replicas、GPU resource limits、nodeSelector
- **Readiness probe**: 等待模型加载完成再接收流量
- **HPA 指标选择**: request_queue_depth > GPU utilization（GPU util 对 LLM 不敏感）
- **Scale-down 谨慎**: stabilization window 防止频繁缩容

参考：vLLM 生产部署文档、K8s HPA 文档。

## C093 — Autoscaling（★★★☆☆）

目标：理解 autoscaling 的决策逻辑——何时 scale、如何避免冷启动、多模型调度。

新增：
```
deploy/
  autoscaling.py   # AutoscalingController
```

核心实现：
```python
# deploy/autoscaling.py
from dataclasses import dataclass
from collections import deque

@dataclass
class ScalingDecision:
    action: str          # "scale_up" | "scale_down" | "noop"
    target_replicas: int
    reason: str

class AutoscalingController:
    """基于请求队列的自动扩缩容。

    决策逻辑:
    1. 监控指标:
       - request_queue_depth: 等待中的请求数
       - avg_ttft: 平均 TTFT (Time To First Token)
       - gpu_utilization: GPU 利用率（辅助参考）

    2. Scale up 条件 (OR):
       - queue_depth > threshold_high 持续 > 30s
       - avg_ttft > sla_target 持续 > 60s

    3. Scale down 条件 (AND):
       - queue_depth < threshold_low 持续 > 300s
       - all_gpu_utilization < target_util 持续 > 300s

    4. 冷启动处理:
       - 新 replica 需要加载模型 (~30s for small, ~300s for large)
       - scale_up 时增加容错余量 (over-provision)
       - 预热: 发送 dummy batch 触发 CUDA Graph capture

    5. 多模型场景:
       - 每个模型独立 autoscaling policy
       - GPU 总量约束: sum(model_i.replicas_i) <= total_gpus
       - 优先级 preemption: 高优模型可抢占低优模型的 GPU
    """
    def __init__(self, config: AutoscalingConfig):
        self.queue_history = deque(maxlen=100)
        self.threshold_high = config.threshold_high       # 默认 10
        self.threshold_low = config.threshold_low          # 默认 2
        self.scale_up_window = config.scale_up_window      # 默认 30s
        self.scale_down_window = config.scale_down_window  # 默认 300s
        self.max_replicas = config.max_replicas
        self.min_replicas = config.min_replicas
        self.cold_start_time = config.cold_start_time      # 模型加载时间

    def evaluate(self, metrics) -> ScalingDecision:
        """评估是否需要扩缩容。"""
        ...

    def _should_scale_up(self, metrics) -> bool:
        """queue_depth > threshold_high 持续 > scale_up_window。"""
        ...

    def _should_scale_down(self, metrics) -> bool:
        """queue_depth < threshold_low 持续 > scale_down_window。"""
        ...
```

核心理解点：
- **Autoscaling 是 Runtime 的"外层控制回路"**：Runtime 不变，外部根据指标调整 replica 数
- **LLM 特有的扩缩容挑战**：模型加载时间长（冷启动）、GPU 量少且贵（不能随意 scale）、TTFT SLA 敏感
- **为什么用 queue_depth 而不是 GPU util**：GPU 利用率在 LLM 推理中天然接近 100%（continuous batching），不是好的扩缩容信号

不做：完整 K8s Custom Metrics API 集成、Prometheus Adapter 配置。
